local function ExecuteAutomatedQuestFarmingRoutine()
    if not AutomatedQuestFarmingEnabled then return end
    if not LocalPlayerReference.Character or not LocalPlayerReference.Character:FindFirstChild("HumanoidRootPart") then RunServiceReference.Heartbeat:wait() return ExecuteAutomatedQuestFarmingRoutine() end
    
    local SelectedToolName = ToolSelectionForFarmingDropdown.Value
    if SelectedToolName and SelectedToolName ~= "" then
        local EquippedTool = LocalPlayerReference.Character:FindFirstChild(SelectedToolName) or LocalPlayerReference.Backpack:FindFirstChild(SelectedToolName)
        
        if not EquippedTool then
            local AvailableToolsList = {"BlackLeg", "Combat", "Eletric", "WaterKungFu"}
            for _, ToolName in pairs(AvailableToolsList) do
                EquippedTool = LocalPlayerReference.Backpack:FindFirstChild(ToolName) or LocalPlayerReference.Character:FindFirstChild(ToolName)
                if EquippedTool then break end
            end
            
            if not EquippedTool then
                for _, ToolContainerObject in pairs({LocalPlayerReference.Backpack, LocalPlayerReference.Character}) do
                    for _, IndividualToolObject in pairs(ToolContainerObject:GetChildren()) do
                        if IndividualToolObject:IsA("Tool") then
                            for _, ToolName in pairs(AvailableToolsList) do
                                if IndividualToolObject.Name:find(ToolName) then
                                    EquippedTool = IndividualToolObject
                                    break
                                end
                            end
                            if EquippedTool then break end
                        end
                    end
                    if EquippedTool then break end
                end
            end
        end
        
        if EquippedTool and EquippedTool:IsA("Tool") and EquippedTool.Parent == LocalPlayerReference.Backpack then 
            LocalPlayerReference.Character.Humanoid:EquipTool(EquippedTool) 
        end
    end
    
    RunServiceReference.Heartbeat:wait() return ExecuteAutomatedQuestFarmingRoutine()
end

local function RetrieveAvailableCharacterTools()
    if not LocalPlayerReference.Character then return {} end
    local AvailableToolsList = {}
    local ValidToolNames = {"BlackLeg", "Combat", "Eletric", "WaterKungFu"}
    
    for _, ToolContainerObject in pairs({LocalPlayerReference.Backpack, LocalPlayerReference.Character}) do
        for _, IndividualToolObject in pairs(ToolContainerObject:GetChildren()) do
            if IndividualToolObject:IsA("Tool") then
                for _, ValidToolName in pairs(ValidToolNames) do
                    if IndividualToolObject.Name:find(ValidToolName) then
                        local alreadyExists = false
                        for _, ExistingTool in pairs(AvailableToolsList) do
                            if ExistingTool == IndividualToolObject.Name then
                                alreadyExists = true
                                break
                            end
                        end
                        if not alreadyExists then
                            table.insert(AvailableToolsList, IndividualToolObject.Name)
                        end
                    end
                end
            end
        end
    end
    return AvailableToolsList
end
